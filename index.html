function openHistory() { const modal = document.getElementById('historyModal'); const list = document.getElementById('historyList'); list.innerHTML = ''; let history = JSON.parse(localStorage.getItem('weeklyHistory') || '[]'); if (history.length === 0) { list.innerHTML = '

Архив пока пуст (он сохраняет только уход за кожей).

'; } else { for (let i = history.length - 1; i >= 0; i--) { let week = history[i]; let div = document.createElement('div'); div.className = 'history-card'; let totalChecks = 0; for (let key in week.checks) if (week.checks[key]) totalChecks++; // Формируем детальный список по дням let detailsHtml = ''; let hasDetails = false; for (let dayIdx = 0; dayIdx < 7; dayIdx++) { let dayChecks = []; // Проверяем, какие процедуры были выполнены в этот день skinColumns.forEach(col => { if (week.checks && week.checks[`check-${dayIdx}-${col}`]) { dayChecks.push(skinColumnNames[col]); } }); let note = week.notes && week.notes[`note-${dayIdx}`] ? week.notes[`note-${dayIdx}`] : ''; // Если в этот день были галочки или есть заметка — выводим этот день if (dayChecks.length > 0 || note.trim() !== '') { hasDetails = true; detailsHtml += `
${daysOfWeek[dayIdx]}
${dayChecks.length > 0 ? `
✅ Выполнено: ${dayChecks.join(', ')}
` : ''} ${note.trim() !== '' ? `
📝 Заметка: ${note}
` : ''}
`; } } if (!hasDetails) { detailsHtml = '

Нет деталей за эту неделю.

'; } div.innerHTML = `

Неделя от ${week.date}

Всего выполнено процедур: ${totalChecks}
${detailsHtml}
`; list.appendChild(div); } } modal.style.display = "block"; }
